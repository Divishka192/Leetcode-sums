import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();

            int[] a = new int[n];
            int total = 0;

            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                total += a[i];
            }

            int minSum = Integer.MAX_VALUE;

            for (int i = 0; i < n - 1; i++) {
                minSum = Math.min(minSum, a[i] + a[i + 1]);
            }

            int ans = total - minSum / 2;

            System.out.println(ans);
        }

        sc.close();
    }
}
