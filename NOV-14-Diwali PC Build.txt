import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();

        while (T-- > 0) {
            int P = sc.nextInt();
            int D = sc.nextInt();
            int K = sc.nextInt();
            int B = sc.nextInt();

            int total = P + (D * K);

            if (total <= B) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }

        sc.close();
    }
}
