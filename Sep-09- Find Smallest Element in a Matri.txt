4. Find Smallest Element in a Matrix

Problem Statement:
Given a matrix, find the smallest element.

Example:

Input:
2 3
10 25 5
8 40 15

Output:
5

Java Code:

import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int r = sc.nextInt();
        int c = sc.nextInt();

        int[][] a = new int[r][c];

        int min = Integer.MAX_VALUE;

        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                a[i][j] = sc.nextInt();

                if (a[i][j] < min) {
                    min = a[i][j];
                }
            }
        }

        System.out.println(min);
    }
}
