import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
	Scanner sc=new Scanner(System.in);
    int c=sc.nextInt();
    int w=sc.nextInt();
    int m=sc.nextInt();
    int p=sc.nextInt();
    int r=sc.nextInt();
    int ans;
    ans=(c*w)-(m*p);
    if(ans>=r){
        System.out.print("YES");
    }
    else {
        System.out.print("NO");
    }

	}
}
