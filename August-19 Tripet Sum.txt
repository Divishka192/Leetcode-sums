package arrays;
import java.util.Scanner;
public class TripletSum {
public static void main(String[] args) {
	Scanner sc=new Scanner(System.in);
	int n=sc.nextInt();
	int a[]=new int[n];
	for(int i=0;i<n;i++) {
		a[i]=sc.nextInt();
	}
	int target=sc.nextInt();
	int sum=0;
	boolean found=false;
	for(int i=0;i<n-2;i++) {
		for(int j=i+1;j<n-1;j++) {
			for(int k=j+1;k<n;k++) {
				sum =a[i]+a[j]+a[k];
				if(sum==target) {					
					found=true;
					System.out.print("PAIR FOUND");
					break;
			}
				
		}
			if(found) 
				break;
		}
			if(found)
				break;
	
	
	}
	if(!found) {
		System.out.print("Not found");
	}
	
}
}
