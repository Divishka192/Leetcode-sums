Longest Contiguous 1's

import java.io.*;
import java.util.*;

import java.util.Scanner;
public class Main {

    public static void main(String[] args) {

        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
       int[] x=new int[n];
       int count=0;
       int max=0;
       for(int i=0;i<n;i++){
        x[i]=sc.nextInt();
        if(x[i]==1){
       count++;
        }
        else{
            count=0;
        }
        if(count>max){
            max=count;
        }
       }
        System.out.print(max);
       
    }
       }
