Find Sum of Each Column

Problem Statement:
Given a matrix, find the sum of elements in each column.

Example:

Input:
2 3
1 2 3
4 5 6

Output:
5
7
9

Explanation:

Column 1 = 1 + 4 = 5
Column 2 = 2 + 5 = 7
Column 3 = 3 + 6 = 9

Java Code:

import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int r = sc.nextInt();
        int c = sc.nextInt();

        int[][] a = new int[r][c];

        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                a[i][j] = sc.nextInt();
            }
        }

        for (int j = 0; j < c; j++) {
            int sum = 0;

            for (int i = 0; i < r; i++) {
                sum += a[i][j];
            }

            System.out.println(sum);
        }
    }
}
